import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, UpdateCommand } from "@aws-sdk/lib-dynamodb";

const bedrockClient = new BedrockRuntimeClient({ region: "eu-north-1" });
const ddbDocClient = DynamoDBDocumentClient.from(new DynamoDBClient({ region: "eu-north-1" }));

// WORKING MODELS (no inference profiles needed)
const MODELS_TO_TRY = [

const ENTRIES_TABLE = "Entry-4xyf7jwncbaxxhwnay24gogxvq-dev";

export const handler = async (event) => {
  console.log(`EVENT: ${JSON.stringify(event)}`);

  for (const record of event.Records) {
    if (record.eventName === 'INSERT') {
      const newEntry = record.dynamodb.NewImage;
      const entryId = newEntry.id.S;
      const entryContent = newEntry.content.S;

      console.log(`📝 Processing new entry: ${entryId}`);

      let success = false;
      
      for (const modelId of MODELS_TO_TRY) {
        try {
          console.log(`🔄 Trying model: ${modelId}`);
          const aiAnalysis = await invokeBedrockModel(modelId, entryContent);
          await updateDynamoDB(entryId, aiAnalysis);
          console.log(`✅ Success with model: ${modelId}`);
          success = true;
          break;
        } catch (error) {
          console.log(`❌ Failed with ${modelId}:`, error.message);
          continue;
        }
      }

      if (!success) {
        console.log("🔄 All models failed, using default values");
        await setDefaultValues(entryId);
      }
    }
  }
  return { status: 'done' };
};

async function invokeBedrockModel(modelId, content) {
  // COMPREHENSIVE PROMPT - Your original detailed version
  const prompt = `You are an experienced CBT (Cognitive Behavioral Therapy) therapist with expertise in mental health and emotional analysis. 

Analyze this journal entry from a therapeutic perspective and provide comprehensive insights.

JOURNAL ENTRY TO ANALYZE:
"${content}"

YOUR TASK:
Provide a detailed JSON response with these exact fields:

1. moodScore: integer 1-10 (1=very low mood, 10=excellent mood)
2. anxietyScore: integer 1-10 (1=completely calm, 10=extremely anxious)
3. stressScore: integer 1-10 (1=no stress, 10=overwhelming stress)
4. detectedEmotions: array of 2-3 primary emotions (e.g., ["anxious", "overwhelmed", "hopeful"])
5. energyLevel: "High", "Medium", or "Low"
6. moodCategory: "Positive", "Neutral", or "Low" based on overall emotional tone
7. aiInsight: A compassionate, supportive 2-3 sentence reflection that validates feelings and offers perspective
8. recommendedAction: A specific, actionable CBT exercise (format: "Action Name\\nCategory\\nDescription")

IMPORTANT SCORING GUIDELINES:
- moodScore: Use the FULL 1-10 scale with nuance (2,3,4,6,7,8,9 are valid scores)
- anxietyScore: Use the FULL 1-10 scale with nuance
- stressScore: Use the FULL 1-10 scale with nuance
- DO NOT just use 1, 5, 10 - be specific to the content
- A mildly positive entry should be 6-7, not 10
- A somewhat anxious entry should be 6-7, not 10
- Be realistic and nuanced in your assessment

RECOMMENDED ACTION FORMAT EXAMPLES:
"Morning Breathing Routine\\nAnxiety Management\\nTry 5 minutes of deep breathing exercises when you wake up to reduce morning anxiety"
"Gratitude Practice\\nPositive Mindset\\nWrite down three things you're grateful for each evening to improve mood"
"Thought Record\\nCognitive Restructuring\\nChallenge negative thoughts by writing evidence for and against them"
"Progressive Muscle Relaxation\\nStress Relief\\nTense and relax each muscle group systematically to release physical tension"
"Behavioral Activation\\nMood Improvement\\nSchedule one enjoyable activity daily to combat low mood"

ANALYSIS GUIDELINES:
- Be empathetic and non-judgmental
- Focus on patterns and underlying themes
- Suggest practical, evidence-based CBT techniques
- Consider the emotional depth and complexity
- Provide insights that encourage self-reflection and growth

Respond with ONLY the JSON object, no other text or explanations.

Required JSON format:
{
  "moodScore": 7,
  "anxietyScore": 6,
  "stressScore": 5,
  "detectedEmotions": ["specific", "emotions", "here"],
  "energyLevel": "High/Medium/Low",
  "moodCategory": "Positive/Neutral/Low",
  "aiInsight": "supportive insight here",
  "recommendedAction": "Action Name\\nCategory\\nDescription"
}`;

  let body;

  if (modelId.startsWith('deepseek.v3')) {
    // DEEPSEEK MODELS - Messages format
    body = JSON.stringify({
      messages: [
        {
          role: "user",
          content: prompt
        }
      ],
      max_tokens: 1500,
      temperature: 0.7
    });
  } else if (modelId.startsWith('qwen.') || modelId.startsWith('openai.gpt-oss')) {
    // QWEN & GPT OSS MODELS - Messages format
    body = JSON.stringify({
      messages: [
        {
          role: "user",
          content: prompt
        }
      ],
      max_tokens: 1500,
      temperature: 0.7
    });
  }

  const command = new InvokeModelCommand({
    modelId,
    contentType: "application/json",
    accept: "application/json", 
    body,
  });

  console.log(`🤖 Calling ${modelId}...`);
  const apiResponse = await bedrockClient.send(command);
  
  const decodedBody = new TextDecoder().decode(apiResponse.body);
  const responseBody = JSON.parse(decodedBody);
  
  let aiText;
  
  if (modelId.startsWith('deepseek.v3') || modelId.startsWith('qwen.') || modelId.startsWith('openai.gpt-oss')) {
    aiText = responseBody.choices?.[0]?.message?.content || responseBody.output?.text || "";
  }

  console.log("RAW AI RESPONSE:", aiText);

  if (!aiText) {
    throw new Error('Empty response from Bedrock');
  }

  // Extract JSON from response
  const jsonMatch = aiText.match(/\{[\s\S]*\}/);
  if (!jsonMatch) {
    throw new Error('No JSON found in response');
  }

  try {
    const aiJson = JSON.parse(jsonMatch[0]);
    console.log("PARSED AI JSON:", aiJson);
    
    return {
      moodScore: validateScore(aiJson.moodScore),
      anxietyScore: validateScore(aiJson.anxietyScore),
      stressScore: validateScore(aiJson.stressScore || 5),
      detectedEmotions: Array.isArray(aiJson.detectedEmotions) ? aiJson.detectedEmotions.slice(0, 3) : ["neutral"],
      energyLevel: validateEnergyLevel(aiJson.energyLevel),
      moodCategory: validateMoodCategory(aiJson.moodCategory), 
      aiInsight: validateInsight(aiJson.aiInsight),
      location: validateRecommendedAction(aiJson.recommendedAction)
    };
    
  } catch (parseError) {
    console.log('JSON parse error:', parseError);
    throw new Error('Invalid JSON format');
  }
}

// Validation functions
function validateScore(score) {
  if (score === null || score === undefined) return 5;
  if (typeof score === 'string') {
    const cleaned = score.replace(/[^\d.-]/g, '');
    const num = parseInt(cleaned);
    if (!isNaN(num) && num >= 1 && num <= 10) return num;
  }
  if (typeof score === 'number') {
    return Math.max(1, Math.min(10, Math.round(score)));
  }
  return 5;
}

function validateEnergyLevel(level) {
  if (!level) return "Medium";
  const cleanLevel = level.trim();
  if (["High", "Medium", "Low"].includes(cleanLevel)) return cleanLevel;
  return "Medium";
}

function validateMoodCategory(category) {
  if (!category) return "Neutral";
  const cleanCategory = category.trim();
  if (["Positive", "Neutral", "Low"].includes(cleanCategory)) return cleanCategory;
  return "Neutral";
}

function validateInsight(insight) {
  if (!insight || typeof insight !== 'string') {
    return "It's helpful to acknowledge and sit with whatever you're feeling right now. Your awareness is the first step toward understanding.";
  }
  return insight.slice(0, 500);
}

function validateRecommendedAction(action) {
  if (!action || typeof action !== 'string') {
    return "Deep Breathing\\nStress Relief\\nTake 3-5 minutes for deep breathing to center yourself";
  }
  return action.slice(0, 300);
}

async function updateDynamoDB(entryId, aiAnalysis) {
  const updateParams = {
    TableName: ENTRIES_TABLE,
    Key: { id: entryId },
    UpdateExpression: "SET moodScore = :ms, anxietyScore = :as, stressScore = :ss, detectedEmotions = :de, energyLevel = :el, moodCategory = :mc, aiInsight = :ai, #loc = :loc",
    ExpressionAttributeNames: { "#loc": "location" },
    ExpressionAttributeValues: {
      ":ms": aiAnalysis.moodScore, ":as": aiAnalysis.anxietyScore, ":ss": aiAnalysis.stressScore,
      ":de": aiAnalysis.detectedEmotions, ":el": aiAnalysis.energyLevel, ":mc": aiAnalysis.moodCategory,
      ":ai": aiAnalysis.aiInsight, ":loc": aiAnalysis.location
    },
  };
  await ddbDocClient.send(new UpdateCommand(updateParams));
  console.log("✅ Updated DynamoDB with AI analysis");
}

async function setDefaultValues(entryId) {
  const defaultAnalysis = {
    moodScore: 5,
    anxietyScore: 5,
    stressScore: 5,
    detectedEmotions: ["neutral"],
    energyLevel: "Medium",
    moodCategory: "Neutral",
    aiInsight: "It's helpful to acknowledge and sit with whatever you're feeling right now.",
    location: "Deep Breathing\\nStress Relief\\nTake 3-5 minutes for deep breathing to center yourself"
  };
  await updateDynamoDB(entryId, defaultAnalysis);
}
